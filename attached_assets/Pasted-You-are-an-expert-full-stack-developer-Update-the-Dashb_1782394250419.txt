// Get winning products with ad details
const sevenDaysAgo = new Date();
sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);

const winners = await ScrapedAd.aggregate([
  { $match: { scrapedAt: { $gte: sevenDaysAgo } } },
  { 
    $group: { 
      _id: { $toLower: '$headline' }, // group by normalized headline
      advertiserName: { $first: '$advertiserName' },
      daysRunning: { $max: '$daysRunning' },
      spendLevel: { $first: '$spendLevel' },
      platform: { $first: '$platform' },
      totalAds: { $sum: 1 },
      advertisers: { $addToSet: '$advertiserName' }
    }
  },
  { 
    $project: {
      name: '$_id',
      advertiserName: 1,
      daysRunning: 1,
      spendLevel: 1,
      platform: 1,
      totalAds: 1,
      advertiserCount: { $size: '$advertisers' },
      winScore: {
        $min: [
          100,
          {
            $add: [
              { $multiply: [{ $size: '$advertisers' }, 4] },
              { $divide: ['$totalAds', 2] },
              { $cond: [{ $gte: ['$daysRunning', 30] }, 20, 0] },
              { $cond: [{ $eq: ['$spendLevel', 'High'] }, 10, 0] }
            ]
          }
        ]
      }
    }
  },
  { $sort: { winScore: -1 } },
  { $limit: 10 }
]);

// Return winners in dashboard response
return Response.json({
  success: true,
  data: {
    totalProducts,
    adsLast7Days,
    adsToday,
    topCategory,
    citiesActive,
    mostActiveCity,
    topWinners: winners.map(w => ({
      name: w.name || 'Unknown Product',
      advertiserName: w.advertiserName || 'Unknown Advertiser',
      daysRunning: w.daysRunning || 0,
      spendLevel: w.spendLevel || 'N/A',
      platform: w.platform || 'facebook',
      winScore: Math.round(w.winScore || 0),
      totalAds: w.totalAds || 0,
      advertiserCount: w.advertiserCount || 0
    })),
    trendingCategories,
    cityDemand
  }
});
