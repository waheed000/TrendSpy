FEATURE 3: Opportunity Score Calculator
Create: services/opportunityService.js
Calculates opportunity score for each product:

javascript
// opportunityScore = globalPopularity × (1 - localAvailability)
// globalPopularity = (shopifyStores + reviews + rating) normalized
// localAvailability = products found on Daraz/OLX (0-1)

export async function calculateOpportunityScore(productName) {
  const global = await getGlobalStats(productName);
  const local = await getLocalAvailability(productName);
  
  const globalScore = Math.min(100, (global.storeCount * 2) + (global.avgRating * 10));
  const localScore = Math.min(100, local.productCount * 20);
  
  return {
    score: Math.round(globalScore * (1 - localScore/100)),
    gap: localScore < 30 ? "HIGH" : localScore < 60 ? "MEDIUM" : "LOW"
  };
}
Export: getOpportunities(limit=20)
