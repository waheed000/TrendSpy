import InAppNotification from '@/models/InAppNotification';
import User from '@/models/User';
import { sendEmail } from './emailService';
import { getAdBasedWinners } from './adWinningService';

// Create an in-app notification
export async function createInAppNotification(userId, type, title, message, link = null) {
  try {
    await InAppNotification.create({
      userId,
      type,
      title,
      message,
      link,
      read: false
    });
  } catch (error) {
    console.error('Failed to create notification:', error);
  }
}

// Get all notifications for a user
export async function getNotifications(userId, limit = 20, offset = 0) {
  return await InAppNotification.find({ userId })
    .sort({ createdAt: -1 })
    .skip(offset)
    .limit(limit);
}

// Get unread count
export async function getUnreadCount(userId) {
  return await InAppNotification.countDocuments({ userId, read: false });
}

// Mark a single notification as read
export async function markAsRead(notificationId, userId) {
  return await InAppNotification.findOneAndUpdate(
    { _id: notificationId, userId },
    { read: true },
    { new: true }
  );
}

// Mark all notifications as read
export async function markAllAsRead(userId) {
  return await InAppNotification.updateMany(
    { userId, read: false },
    { read: true }
  );
}

// Delete a notification
export async function deleteNotification(notificationId, userId) {
  return await InAppNotification.findOneAndDelete({ _id: notificationId, userId });
}

// Send daily digest email to all users who opted in

// Send notification for winning product (triggered by alert system)
export async function notifyWinningProduct(userId, product) {
  const title = `🔥 New Winning Product: ${product.name}`;
  const message = `Score: ${product.winScore}/100 · ${product.advertiserCount} advertisers · ${product.totalAds} ads`;

3. Create app/api/notifications/route.js
javascript
import { getNotifications, markAsRead, markAllAsRead, deleteNotification, getUnreadCount } from '@/services/notificationService';
import { withAuth } from '@/middleware/auth';

// GET /api/notifications — fetch all notifications
async function getNotificationsHandler(request, { user }) {
  const { searchParams } = new URL(request.url);
  const limit = parseInt(searchParams.get('limit') || '20');
  const offset = parseInt(searchParams.get('offset') || '0');
  
  const notifications = await getNotifications(user._id, limit, offset);
  const unread = await getUnreadCount(user._id);
  
  return Response.json({
    success: true,
    data: { notifications, unread, limit, offset }
  });
}

// PUT /api/notifications — mark as read
async function putNotificationsHandler(request, { user }) {
  const body = await request.json();
  const { notificationId } = body;
  
  if (notificationId === 'all') {
    await markAllAsRead(user._id);
  } else if (notificationId) {
    await markAsRead(notificationId, user._id);
  }
  
  const unread = await getUnreadCount(user._id);
  return Response.json({ success: true, unread });
}

// DELETE /api/notifications — delete a notification
async function deleteNotificationsHandler(request, { user }) {
  const { searchParams } = new URL(request.url);
  const notificationId = searchParams.get('id');
  
  if (!notificationId) {
    return Response.json({ success: false, error: 'Notification ID required' }, { status: 400 });
  }
  
  await deleteNotification(notificationId, user._id);
  return Response.json({ success: true });
}

export const GET = withAuth(getNotificationsHandler);
export const PUT = withAuth(putNotificationsHandler);
export const DELETE = withAuth(deleteNotificationsHandler);

import { useState, useEffect } from 'react';
import { Link } from 'react-router-dom';
import { FiBell } from 'react-icons/fi';
import { useAuthStore } from '@/store/useAuthStore';

export default function NotificationBell() {
  const { token } = useAuthStore();
  const [unread, setUnread] = useState(0);
  const [open, setOpen] = useState(false);
  
  useEffect(() => {
    if (token) {
      fetchUnreadCount();
      // Poll every 30 seconds
      const interval = setInterval(fetchUnreadCount, 30000);
      return () => clearInterval(interval);
    }
  }, [token]);
  
  const fetchUnreadCount = async () => {
    try {
      const res = await fetch('/api/notifications/count', {
        headers: { 'Authorization': `Bearer ${token}` }
      });
      const data = await res.json();
      if (data.success) setUnread(data.unread);
    } catch (error) {
      console.error('Failed to fetch unread count:', error);
    }
  };
  
  return (
    <div className="relative">
      <button 
        onClick={() => setOpen(!open)}
        className="relative p-2 rounded-full hover:bg-gray-100 dark:hover:bg-gray-700 transition"
      >
        <FiBell className="text-xl text-gray-600 dark:text-gray-300" />
        {unread > 0 && (
          <span className="absolute -top-1 -right-1 bg-red-500 text-white text-xs w-5 h-5 rounded-full flex items-center justify-center">
            {unread > 9 ? '9+' : unread}
          </span>
        )}
      </button>
      
      {open && (
        <div className="absolute right-0 mt-2 w-80 bg-white dark:bg-gray-800 rounded-lg shadow-lg border border-gray-200 dark:border-gray-700 z-50 max-h-96 overflow-y-auto">
          <div className="p-3 border-b border-gray-200 dark:border-gray-700">
            <div className="flex justify-between items-center">
              <span className="font-semibold text-gray-900 dark:text-white">Notifications</span>
              {unread > 0 && (
                <button 
                  onClick={async () => {
                    await fetch('/api/notifications', {
                      method: 'PUT',
                      headers: { 
                        'Content-Type': 'application/json',
                        'Authorization': `Bearer ${token}` 
                      },
                      body: JSON.stringify({ notificationId: 'all' })
                    });
                    setUnread(0);
                  }}
                  className="text-xs text-teal-600 dark:text-teal-400 hover:underline"

                  Mark all read

8. Create src/pages/Notifications.jsx
jsx
import { useState, useEffect } from 'react';
import { useAuthStore } from '@/store/useAuthStore';
import { Link } from 'react-router-dom';

export default function Notifications() {
  const { token } = useAuthStore();
  const [notifications, setNotifications] = useState([]);
  const [loading, setLoading] = useState(true);
  const [unread, setUnread] = useState(0);
  
  useEffect(() => {
    fetchNotifications();
  }, []);
  
  const fetchNotifications = async () => {
    try {
      const res = await fetch('/api/notifications', {
        headers: { 'Authorization': `Bearer ${token}` }
      });
      const data = await res.json();
      if (data.success) {
        setNotifications(data.data.notifications);
        setUnread(data.data.unread);
      }
    } catch (error) {
      console.error('Failed to fetch notifications:', error);
    } finally {
      setLoading(false);
    }
  };
  
  const markAsRead = async (id) => {
    try {
      await fetch('/api/notifications', {
        method: 'PUT',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`
        },
        body: JSON.stringify({ notificationId: id })
      });
      setNotifications(prev => prev.map(n => 
        n._id === id ? { ...n, read: true } : n
      ));
      setUnread(prev => Math.max(0, prev - 1));
    } catch (error) {
      console.error('Failed to mark as read:', error);
    }
  };
  
  const markAllAsRead = async () => {
    try {
      await fetch('/api/notifications', {
        method: 'PUT',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`
        },
        body: JSON.stringify({ notificationId: 'all' })
      });
      setNotifications(prev => prev.map(n => ({ ...n, read: true })));
      setUnread(0);
    } catch (error) {
      console.error('Failed to mark all as read:', error);
    }
  };
  
  if (loading) {
    return (
      <div className="min-h-screen bg-gray-50 dark:bg-gray-900 p-6 flex items-center justify-center">
        <div className="text-gray-500">Loading notifications...</div>
      </div>
    );
  }
  
  return (
    <div className="min-h-screen bg-gray-50 dark:bg-gray-900 p-6">
      <div className="max-w-3xl mx-auto">
        <div className="flex justify-between items-center mb-6">
          <h1 className="text-2xl font-bold text-gray-900 dark:text-white">
            🔔 Notifications
            {unread > 0 && (
              <span className="ml-2 text-sm bg-red-500 text-white px-2 py-1 rounded-full">
                {unread} unread
              </span>
            )}
          </h1>
          {unread > 0 && (
            <button
              onClick={markAllAsRead}
              className="text-sm text-teal-600 dark:text-teal-400 hover:underline"
            >
              Mark all as read
            </button>
          )}
        </div>
        
        {notifications.length === 0 ? (
          <div className="bg-white dark:bg-gray-800 rounded-lg p-8 text-center border border-gray-200 dark:border-gray-700">
            <p className="text-gray-500 dark:text-gray-400">No notifications yet.</p>
            <p className="text-sm text-gray-400 dark:text-gray-500 mt-1">
              We'll notify you when new winning products are found.
            </p>
          </div>
        ) : (
          <div className="space-y-3">
            {notifications.map((notification) => (
              <div
                key={notification._id}
                className={`bg-white dark:bg-gray-800 rounded-lg p-4 border ${
                  notification.read 
                    ? 'border-gray-200 dark:border-gray-700' 
                    : 'border-teal-200 dark:border-teal-700 bg-teal-50/30 dark:bg-teal-900/10'
                }`}
                onClick={() => {
                  if (!notification.read) markAsRead(notification._id);
                  if (notification.link) window.location.href = notification.link;
                }}
              >
                <div className="flex items-start justify-between">
                  <div>
                    <div className="flex items-center gap-2">
                      <span className="font-medium text-gray-900 dark:text-white">
                        {notification.title}
                      </span>
                      {!notification.read && (
                        <span className="w-2 h-2 bg-teal-500 rounded-full"></span>
                      )}
                    </div>
                    <p className="text-sm text-gray-600 dark:text-gray-300 mt-1">
                      {notification.message}
                    </p>
                    <p className="text-xs text-gray-400 dark:text-gray-500 mt-2">
                      {new Date(notification.createdAt).toLocaleString()}
                    </p>
                  </div>
                  {!notification.read && (
                    <button
                      onClick={(e) => {
                        e.stopPropagation();
                        markAsRead(notification._id);
                      }}
                      className="text-xs text-gray-400 hover:text-gray-600 dark:hover:text-gray-300"
                    >
                      Mark read
                    </button>
                  )}
                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
