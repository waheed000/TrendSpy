import ScrapedAd from '@/models/ScrapedAd';
import Product from '@/models/Product';

// Extract product name from ad headline
function extractProductName(headline) {
  // Remove common ad phrases
  const cleaned = headline
    .replace(/limited time|sale|offer|buy now|shop now|free shipping|order now/gi, '')
    .replace(/[^\w\s]/g, '')
    .trim();
  
  // Take first 3-5 words as product name
  return cleaned.split(' ').slice(0, 4).join(' ');
}

// Get winning products from Facebook Ads
export async function getWinningProductsFromFBAds(limit = 20) {
  const sevenDaysAgo = new Date();
  sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);
  
  // Group ads by product name
  const ads = await ScrapedAd.find({
    scrapedAt: { $gte: sevenDaysAgo }
  });
  
  // Group by extracted product name
  const productMap = new Map();
  
  for (const ad of ads) {
    const productName = extractProductName(ad.headline || '');
    if (!productName) continue;
    
    if (!productMap.has(productName)) {
      productMap.set(productName, {
        name: productName,
        advertiserCount: new Set(),
        totalAds: 0,
        maxDaysRunning: 0,
        platforms: new Set(),
        spendLevels: []
      });
    }
    
    const entry = productMap.get(productName);
    if (ad.advertiserName) entry.advertiserCount.add(ad.advertiserName);
    entry.totalAds++;
    if (ad.daysRunning) entry.maxDaysRunning = Math.max(entry.maxDaysRunning, ad.daysRunning);
    if (ad.platform) entry.platforms.add(ad.platform);
    if (ad.spendLevel) entry.spendLevels.push(ad.spendLevel);
  }
  
  // Calculate win score for each product
  const results = [];
  for (const [name, data] of productMap) {
    let score = 0;
    
    // More advertisers = higher score (max 40 points)
    score += Math.min(40, data.advertiserCount.size * 8);
    
    // More total ads = higher score (max 30 points)
    score += Math.min(30, data.totalAds);
    
    // Ads running 30+ days = proven winner (20 points)
    if (data.maxDaysRunning >= 30) score += 20;
    else if (data.maxDaysRunning >= 14) score += 10;
    
    // High spend = high confidence (10 points)
    const highSpendCount = data.spendLevels.filter(s => s === 'High').length;
    if (highSpendCount > 0) score += Math.min(10, highSpendCount * 2);
    
    results.push({
      name,
      winScore: Math.min(100, Math.round(score)),
      advertiserCount: data.advertiserCount.size,
      totalAds: data.totalAds,
      maxDaysRunning: data.maxDaysRunning,
      platforms: Array.from(data.platforms),
      source: 'facebook_ads'
    });
  }
  
  return results.sort((a, b) => b.winScore - a.winScore).slice(0, limit);
}

Create app/api/products/winning/route.js:

javascript
import { getWinningProducts } from '@/services/adWinningService';

export async function GET() {
  try {
    const winners = await getWinningProducts(20);
    
    return Response.json({
      success: true,
      data: {
        products: winners,
        total: winners.length,
        source: 'facebook_ads (real-time)',
        lastUpdated: new Date().toISOString()
      }
    });
  } catch (error) {
    console.error('Error fetching winning products:', error);
    return Response.json(
      { success: false, error: error.message },
      { status: 500 }
    );
  }
}
