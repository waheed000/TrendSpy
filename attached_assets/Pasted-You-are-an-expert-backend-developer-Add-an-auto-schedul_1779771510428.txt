import cron from 'node-cron';
import { scrapeFacebookAds } from '@/scrapers/fbAdsScraper';
import { darazScraper } from '@/scrapers/darazScraper';
import { olxScraper } from '@/scrapers/olxScraper';

// Track last run status
let lastRunStatus = {
  facebookAds: null,
  daraz: null,
  olx: null,
  googleTrends: null,
  tiktok: null,
  news: null,
  suppliers: null
};

// Categories to scrape for Facebook Ads
const AD_CATEGORIES = ['Electronics', 'Fashion', 'Beauty', 'Home', 'Sports'];

// Run Facebook Ads scraper
async function runFacebookAdsScraper() {
  console.log(`[${new Date().toISOString()}] Starting Facebook Ads scraper...`);
  let totalSaved = 0;
  
  for (const category of AD_CATEGORIES) {
    try {
      const result = await scrapeFacebookAds(category, category);
      totalSaved += result.savedNew || 0;
      console.log(`  ✅ ${category}: ${result.savedNew || 0} new ads`);
      await new Promise(resolve => setTimeout(resolve, 5000)); // 5 sec delay
    } catch (error) {
      console.error(`  ❌ ${category} failed:`, error.message);
    }
  }
  
  lastRunStatus.facebookAds = { completedAt: new Date(), saved: totalSaved };
  console.log(`✅ Facebook Ads scraper done. Total new ads: ${totalSaved}`);
}

// Run Daraz scraper
async function runDarazScraper() {

export function startAllScrapers() {

2. Create app/api/scraper/status/route.js
javascript
import { lastRunStatus } from '@/lib/scrapeScheduler';

export async function GET() {
  return Response.json({
    success: true,
    status: lastRunStatus,
    nextRuns: {
      facebookAds: getNextCronTime('0 */6 * * *'),
      darazOlx: getNextCronTime('0 2,14 * * *'),
      dailyScrapers: getNextCronTime('0 3 * * *'),
      suppliers: getNextCronTime('0 4 * * 0')
    }
  });
}

function getNextCronTime(cronExpression) {
  // Simple parser for display
  const now = new Date();
  const next = new Date(now);
  
  if (cronExpression.includes('*/6')) {
    next.setHours(Math.ceil(now.getHours() / 6) * 6, 0, 0, 0);
  } else if (cronExpression.includes('2,14')) {
    const hours = now.getHours();
    if (hours < 2) next.setHours(2, 0, 0, 0);
    else if (hours < 14) next.setHours(14, 0, 0, 0);
    else next.setDate(next.getDate() + 1);
  } else if (cronExpression.includes('3')) {
    next.setDate(next.getDate() + 1);
    next.setHours(3, 0, 0, 0);
  }
  
  return next.toISOString();
}

import { scrapeFacebookAds } from '@/scrapers/fbAdsScraper';

export async function POST(request) {

  switch(scraper) {
    case 'facebook':
      const result = await scrapeFacebookAds(category, category);
      return Response.json({ success: true, result });
