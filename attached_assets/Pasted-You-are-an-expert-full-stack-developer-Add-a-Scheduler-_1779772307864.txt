  // Get next run times

    facebookAds: getNextCronTime('0 */6 * * *'),
    darazOlx: getNextCronTime('0 2,14 * * *'),

  const productCount = await Product.countDocuments();

      totalProducts: productCount,
      totalSuppliers: supplierCount
    },
    environment: {

    }
  });

Create src/pages/admin/SchedulerDashboard.jsx
jsx
import { useState, useEffect } from 'react';
import { useAuthStore } from '@/store/useAuthStore';

export default function SchedulerDashboard() {
  const [status, setStatus] = useState(null);
  const [loading, setLoading] = useState(true);
  const [triggering, setTriggering] = useState(null);
  const { token } = useAuthStore();
  
  useEffect(() => {
    fetchStatus();
    // Refresh every 30 seconds
    const interval = setInterval(fetchStatus, 30000);
    return () => clearInterval(interval);
  }, []);
  
  const fetchStatus = async () => {
    try {
      const res = await fetch('/api/scraper/status');
      const data = await res.json();
      setStatus(data);
    } catch (error) {
      console.error('Failed to fetch status:', error);
    } finally {
      setLoading(false);
    }
  };
  
  const triggerScraper = async (scraperName) => {
    setTriggering(scraperName);
    try {
      const res = await fetch('/api/scraper/trigger', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`
        },
        body: JSON.stringify({ scraper: scraperName })
      });
      const data = await res.json();
      if (data.success) {
        // Refresh status after 5 seconds
        setTimeout(fetchStatus, 5000);
      }
    } catch (error) {
      console.error('Failed to trigger:', error);
    } finally {
      setTriggering(null);
    }
  };
  
  if (loading) {
    return (
      <div className="min-h-screen bg-gray-900 flex items-center justify-center">
        <div className="text-white text-xl">Loading scheduler status...</div>
      </div>
    );
  }
  
  const scrapers = [
    { 
      key: 'facebookAds', 
      name: 'Facebook Ads', 
      icon: '📱', 
      color: 'blue',
      schedule: 'Every 6 hours',
      nextRun: status?.nextRuns?.facebookAds
    },
    { 
      key: 'darazOlx', 
      name: 'Daraz + OLX', 
      icon: '🛒', 
      color: 'orange',
      schedule: 'Every 12 hours (2AM/2PM)',
      nextRun: status?.nextRuns?.darazOlx
    },
    { 
      key: 'googleTrends', 
      name: 'Google Trends', 
      icon: '📈', 
      color: 'green',
      schedule: 'Daily at 3AM',
      nextRun: status?.nextRuns?.googleTrends
    },
    { 
      key: 'tiktok', 
      name: 'TikTok', 
      icon: '🎵', 
      color: 'pink',
      schedule: 'Daily at 3AM',
      nextRun: status?.nextRuns?.tiktok
    },
    { 
      key: 'news', 
      name: 'News', 
      icon: '📰', 
      color: 'gray',
      schedule: 'Daily at 3AM',
      nextRun: status?.nextRuns?.news
    },
    { 
      key: 'suppliers', 
      name: 'Suppliers', 
      icon: '🏭', 
      color: 'purple',
      schedule: 'Weekly on Sunday 4AM',
      nextRun: status?.nextRuns?.suppliers
    }
  ];
  
  return (
    <div className="min-h-screen bg-gray-900 py-8 px-4">
      <div className="max-w-7xl mx-auto">
        {/* Header */}
        <div className="mb-8">
          <h1 className="text-3xl font-bold text-white mb-2">🤖 Scraper Scheduler Dashboard</h1>
          <p className="text-gray-400">Monitor and control all automatic scrapers</p>
        </div>
        
        {/* Stats Cards */}
        <div className="grid grid-cols-1 md:grid-cols-3 gap-6 mb-8">
          <div className="bg-gray-800 rounded-lg p-6 border border-gray-700">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-gray-400 text-sm">Total Products</p>
                <p className="text-2xl font-bold text-white">{status?.stats?.totalProducts || 0}</p>
              </div>
              <span className="text-3xl">📦</span>
            </div>
          </div>
          <div className="bg-gray-800 rounded-lg p-6 border border-gray-700">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-gray-400 text-sm">Total Ads</p>
                <p className="text-2xl font-bold text-white">{status?.stats?.totalAds || 0}</p>
              </div>
              <span className="text-3xl">📢</span>
            </div>
          </div>
          <div className="bg-gray-800 rounded-lg p-6 border border-gray-700">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-gray-400 text-sm">Total Suppliers</p>
                <p className="text-2xl font-bold text-white">{status?.stats?.totalSuppliers || 0}</p>
              </div>
              <span className="text-3xl">🏭</span>
            </div>
          </div>
        </div>
        
        {/* Auto Scraper Status */}
        <div className="bg-gray-800 rounded-lg p-4 mb-6 flex items-center justify-between">
          <div className="flex items-center gap-3">
            <div className={`w-3 h-3 rounded-full ${status?.environment?.autoScraperEnabled ? 'bg-green-500' : 'bg-red-500'}`}></div>
            <span className="text-white">Auto Scraper</span>
          </div>
          <span className={`text-sm ${status?.environment?.autoScraperEnabled ? 'text-green-400' : 'text-red-400'}`}>
            {status?.environment?.autoScraperEnabled ? 'Enabled' : 'Disabled'}
          </span>
        </div>
        
        {/* Scrapers Grid */}
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {scrapers.map((scraper) => {
            const runData = status?.scrapers?.[scraper.key];
            const isRunning = runData?.status === 'running';
            const lastRun = runData?.completedAt;
            const lastResult = runData?.saved || runData?.products || runData?.listings;
            
            return (
              <div key={scraper.key} className="bg-gray-800 rounded-lg border border-gray-700 overflow-hidden">
                <div className="p-4 border-b border-gray-700">
                  <div className="flex items-center justify-between">
                    <div className="flex items-center gap-2">
                      <span className="text-2xl">{scraper.icon}</span>
                      <h3 className="text-lg font-semibold text-white">{scraper.name}</h3>
                    </div>
                    <div className={`w-2 h-2 rounded-full ${isRunning ? 'bg-yellow-500 animate-pulse' : 'bg-green-500'}`}></div>
                  </div>
                  <p className="text-xs text-gray-500 mt-1">{scraper.schedule}</p>
                </div>
                
                <div className="p-4 space-y-2">
                  <div className="flex justify-between text-sm">
                    <span className="text-gray-400">Last Run:</span>
                    <span className="text-gray-300">
                      {lastRun ? new Date(lastRun).toLocaleString() : 'Never'}
                    </span>
                  </div>
                  <div className="flex justify-between text-sm">
                    <span className="text-gray-400">Next Run:</span>
                    <span className="text-gray-300">
                      {scraper.nextRun ? new Date(scraper.nextRun).toLocaleString() : 'Not scheduled'}
                    </span>
                  </div>
                  {lastResult !== undefined && (
                    <div className="flex justify-between text-sm">
                      <span className="text-gray-400">Last Result:</span>
                      <span className="text-green-400">+{lastResult} items</span>
                    </div>
                  )}
                  {runData?.error && (
                    <div className="text-red-400 text-xs mt-1">{runData.error}</div>
                  )}
                  
                  <button
                    onClick={() => triggerScraper(scraper.key)}
                    disabled={triggering === scraper.key}
                    className="w-full mt-3 bg-gray-700 hover:bg-gray-600 text-white text-sm py-2 rounded-lg transition disabled:opacity-50"
                  >
                    {triggering === scraper.key ? 'Triggering...' : '▶ Run Now'}
                  </button>
                </div>
              </div>
            );
          })}
        </div>
        
        {/* Refresh Button */}
        <div className="text-center mt-8">
          <button
            onClick={fetchStatus}
            className="bg-blue-600 hover:bg-blue-700 text-white px-6 py-2 rounded-lg transition"
          >
            🔄 Refresh Status
          </button>
        </div>
      </div>
    </div>
  );
}
