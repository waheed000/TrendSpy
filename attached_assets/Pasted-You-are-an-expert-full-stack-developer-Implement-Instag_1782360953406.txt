6. Update services/groqService.js
Permanent implementation with fallback:

javascript
import Groq from 'groq-sdk';

// Initialize Groq client only if API key exists
let groqClient = null;
const GROQ_API_KEY = process.env.GROQ_API_KEY;

if (GROQ_API_KEY && GROQ_API_KEY !== 'your_groq_api_key_here') {
  try {
    groqClient = new Groq({ apiKey: GROQ_API_KEY });
    console.log('✅ Groq AI initialized with real API key');
  } catch (error) {
    console.log('⚠️ Groq initialization failed:', error.message);
  }
} else {
  console.log('⚠️ GROQ_API_KEY not set. Using local fallback for AI analysis.');
}

export async function analyzeProduct(productName, productData = {}) {
  // Try real Groq API first
  if (groqClient) {
    try {
      const response = await groqClient.chat.completions.create({
        model: 'llama-3.3-70b-versatile',
        messages: [
          { role: 'system', content: 'You are an expert e-commerce analyst for Pakistani sellers. Analyze products for profitability, ad copy, and sourcing.' },
          { role: 'user', content: `Analyze this product for Pakistan market: ${productName}. Give profit margin, ad copy in English and Urdu, and supplier suggestions.` }
        ],
        temperature: 0.7,
        max_tokens: 1024,
      });
      
      return {
        success: true,
        source: 'groq',
        analysis: response.choices[0].message.content,
        profitCalculator: extractProfitFromText(response.choices[0].message.content),
        adCopy: extractAdCopyFromText(response.choices[0].message.content),
        suppliers: extractSuppliersFromText(response.choices[0].message.content)
      };
    } catch (error) {
      console.error('Groq API error:', error.message);
      // Fall through to local fallback
    }
  }
  
  // Local fallback (always works, no API key needed)
  console.log('🔄 Using local AI fallback for:', productName);
  return generateLocalAnalysis(productName, productData);
}

// Local fallback function (never fails)
