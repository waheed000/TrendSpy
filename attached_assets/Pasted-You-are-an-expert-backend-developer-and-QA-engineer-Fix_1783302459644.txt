#### 1. Update `scrapers/fbAdsScraper.js`

Replace fake adId extraction with REAL extraction:

```javascript
// Extract REAL adId from the page's internal JSON
function extractRealAdId(page) {
  // Method 1: Extract from GraphQL response in page HTML
  const html = await page.content();
  const match = html.match(/"ad_archive_id":"(\d+)"/);
  if (match && match[1]) {
    return match[1];
  }
  
  // Method 2: Extract from data attribute (numeric only)
  const adId = await page.evaluate(() => {
    const el = document.querySelector('[data-ad-id]');
    if (el) {
      const id = el.getAttribute('data-ad-id');
      if (id && /^[0-9]{10,}$/.test(id)) {
        return id;
      }
    }
    return null;
  });
  
  return adId;
}

// Validate before saving
function isValidAdId(adId) {
  return adId && /^[0-9]{10,}$/.test(adId);
}
