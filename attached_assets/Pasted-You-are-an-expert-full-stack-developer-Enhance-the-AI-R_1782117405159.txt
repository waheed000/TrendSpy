3. Pass Inventory Advice Logic
In the AI Report handler, calculate inventory advice based on:

daysRunning (30+ days → 100 units, 14+ days → 50 units, <14 days → 30 units)

advertiserCount (more advertisers → higher demand → more stock)

javascript
function getInventoryAdvice(daysRunning, advertiserCount) {
  let recommendedOrder = 30;
  if (daysRunning >= 30) recommendedOrder = 100;
  else if (daysRunning >= 14) recommendedOrder = 50;
  
  if (advertiserCount > 20) recommendedOrder = Math.min(recommendedOrder * 1.5, 200);
  
  return {
    recommendedOrder: `${Math.round(recommendedOrder)} units`,
    reorderWhen: `${Math.round(recommendedOrder * 0.2)} units left`,
    bulkDiscount: recommendedOrder > 100 ? 'Order 100+ units for 15% off' : 'Start with small batch'
  };
}
