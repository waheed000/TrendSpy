import rateLimit from 'express-rate-limit';

// General API rate limit — 100 requests per 15 minutes
export const apiLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 100,
  message: 'Too many requests, please try again later.',
  standardHeaders: true,
  legacyHeaders: false,
});

// Auth rate limit — 10 attempts per hour
export const authLimiter = rateLimit({
  windowMs: 60 * 60 * 1000,
  max: 10,
  message: 'Too many login attempts, please try again later.',
  standardHeaders: true,
  legacyHeaders: false,
});

// Scraper rate limit — 5 triggers per hour
export const scraperLimiter = rateLimit({
  windowMs: 60 * 60 * 1000,
  max: 5,
  message: 'Too many scrape requests, please try again later.',
  standardHeaders: true,
  legacyHeaders: false,

3. Create middleware/logger.js
javascript
export function logRequest(req, res, next) {
  const start = Date.now();
  
  res.on('finish', () => {
    const duration = Date.now() - start;
    const log = {
      timestamp: new Date().toISOString(),
      method: req.method,
      url: req.url,
      status: res.statusCode,
      duration: `${duration}ms`,
      ip: req.ip || req.headers['x-forwarded-for'] || 'unknown',
      userAgent: req.headers['user-agent'] || 'unknown'
    };
    console.log(JSON.stringify(log));
  });
  
  next();
}

export function logError(err, req, res, next) {
  console.error(JSON.stringify({
    timestamp: new Date().toISOString(),
    error: err.message,
    stack: err.stack,
    url: req.url,
    method: req.method,
    ip: req.ip
  }));
  
  res.status(500).json({ 
    success: false, 
    error: 'Internal server error' 
  });
}
4. Apply Middleware in app/api/[...route]/middleware.js or individual routes
javascript
import { apiLimiter, authLimiter, scraperLimiter } from '@/middleware/rateLimit';
import { logRequest, logError } from '@/middleware/logger';

// Apply logging to all routes
app.use(logRequest);

// Apply rate limits
app.use('/api/auth/login', authLimiter);
app.use('/api/auth/register', authLimiter);
app.use('/api/ads/refresh', scraperLimiter);
app.use('/api/suppliers/scrape', scraperLimiter);
app.use('/api/', apiLimiter);

// Error handling
app.use(logError);
5. Create app/api/user/export/route.js (GDPR Data Export)
javascript
import { withAuth } from '@/middleware/auth';
import { connectDB } from '@/lib/db';
import User from '@/models/User';
import Alert from '@/models/Alert';
import AlertLog from '@/models/AlertLog';
import InAppNotification from '@/models/InAppNotification';

async function exportData(request, { user }) {
  await connectDB();
  
  const [alerts, alertLogs, notifications] = await Promise.all([
    Alert.find({ userId: user._id }),
    AlertLog.find({ userId: user._id }),
    InAppNotification.find({ userId: user._id })
  ]);
  
  const exportData = {
    user: {
      name: user.name,
      email: user.email,
      phoneNumber: user.phoneNumber,
      city: user.selectedCity || '',
      categories: user.selectedCategories || [],
      platforms: user.selectedPlatforms || [],
      emailNotifications: user.emailNotifications,
      whatsappNotifications: user.whatsappNotifications,
      dailyDigest: user.dailyDigest,
      subscriptionPlan: user.subscriptionPlan,
      createdAt: user.createdAt
    },
    alerts: alerts.map(a => ({
      city: a.city,
      category: a.category,
      minWinScore: a.minWinScore,
      channel: a.channel,
      isActive: a.isActive,
      triggerCount: a.triggerCount,
      createdAt: a.createdAt
    })),
    alertHistory: alertLogs.map(l => ({
      productName: l.productName,
      winScore: l.winScore,
      channel: l.channel,
      delivered: l.delivered,
      sentAt: l.sentAt
    })),
    notifications: notifications.map(n => ({
      type: n.type,
      title: n.title,
      message: n.message,
      read: n.read,
      createdAt: n.createdAt
    })),
    exportedAt: new Date().toISOString()
  };
  
  return Response.json({ success: true, data: exportData });
}

export const GET = withAuth(exportData);
6. Create app/api/user/delete/route.js (GDPR Data Deletion)
javascript
import { withAuth } from '@/middleware/auth';
import { connectDB } from '@/lib/db';
import User from '@/models/User';
import Alert from '@/models/Alert';
import AlertLog from '@/models/AlertLog';
import InAppNotification from '@/models/InAppNotification';

async function deleteAccount(request, { user }) {
  await connectDB();
  
  await Promise.all([
    Alert.deleteMany({ userId: user._id }),
    AlertLog.deleteMany({ userId: user._id }),
    InAppNotification.deleteMany({ userId: user._id })
  ]);
  
  await User.findByIdAndDelete(user._id);
  
  return Response.json({
    success: true,
    message: 'Account and all data permanently deleted.'
  });
}

export const DELETE = withAuth(deleteAccount);
