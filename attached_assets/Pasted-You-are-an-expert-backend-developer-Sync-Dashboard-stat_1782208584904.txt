```javascript
import { connectDB } from '@/lib/db';
import Product from '@/models/Product';
import ScrapedAd from '@/models/ScrapedAd';
import { getAdBasedWinners } from '@/services/adWinningService';

export async function GET() {
  try {
    await connectDB();
    
    // 1. Total Products Tracked
    const totalProducts = await Product.countDocuments();
    
    // 2. Ads in last 7 days
    const sevenDaysAgo = new Date();
    sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);
    
    const adsLast7Days = await ScrapedAd.countDocuments({
      scrapedAt: { $gte: sevenDaysAgo }
    });
    
    // 3. Ads Scraped Today
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const tomorrow = new Date(today);
    tomorrow.setDate(tomorrow.getDate() + 1);
    
    const adsToday = await ScrapedAd.countDocuments({
      scrapedAt: { $gte: today, $lt: tomorrow }
    });
    
    // 4. Top Category (by ad count + unique advertisers)
    const categoryStats = await ScrapedAd.aggregate([
      { $match: { scrapedAt: { $gte: sevenDaysAgo } } },
      { 
        $group: { 
          _id: '$category', 
          count: { $sum: 1 },
          advertisers: { $addToSet: '$advertiserName' }
        } 
      },
      { 
        $project: { 
          name: '$_id', 
          count: 1, 
          advertisers: { $size: '$advertisers' } 
        } 
      },
      { $sort: { count: -1 } },
      { $limit: 1 }
    ]);
    
    const topCategory = categoryStats.length > 0 ? {
      name: categoryStats[0].name || 'Uncategorized',
      count: categoryStats[0].count,
      advertisers: categoryStats[0].advertisers
    } : null;
    
    // 5. City Demand
    const cityDemand = await ScrapedAd.aggregate([
      { $match: { city: { $exists: true, $ne: null } } },
      { $group: { _id: '$city', count: { $sum: 1 } } },
      { $sort: { count: -1 } }
    ]);
    
    const citiesActive = cityDemand.length;
    const mostActiveCity = cityDemand.length > 0 ? cityDemand[0]._id : 'N/A';
    
    // 6. Top Products (same as Product Hunt winners)
    const winners = await getAdBasedWinners(5);
    
    // Map to dashboard format
    const topProducts = winners.map((p, idx) => ({
      rank: idx + 1,
      name: p.name,
      category: p.category || 'N/A',
      winScore: p.winScore || 0
    }));
    
    // 7. Trending Categories (all categories with counts)
    const trendingCategories = await ScrapedAd.aggregate([
      { $match: { scrapedAt: { $gte: sevenDaysAgo } } },
      { $group: { _id: '$category', count: { $sum: 1 } } },
      { $sort: { count: -1 } },
      { $limit: 5 }
    ]);
    
    // 8. City Demand (detailed for display)
    const cityDemandDetailed = cityDemand.map(c => ({
      city: c._id || 'Unknown',
      count: c.count
    }));
    
    return Response.json({
      success: true,
      data: {
        totalProducts,
        adsLast7Days,
        adsToday,
        topCategory,
        citiesActive,
        mostActiveCity,
        topProducts,
        trendingCategories: trendingCategories.map(c => ({
          name: c._id || 'Uncategorized',
          count: c.count
        })),
        cityDemand: cityDemandDetailed
      }
    });
    
  } catch (error) {
    console.error('Dashboard stats error:', error);
    return Response.json(
      { success: false, error: error.message },
      { status: 500 }
    );
  }
}
