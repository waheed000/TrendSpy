```javascript
import jwt from 'jsonwebtoken';
import { User } from '@/models';

export async function authMiddleware(request) {
  try {
    // Try to get token from cookies (Next.js cookies)
    let token = request.cookies?.get('token')?.value;
    
    // Try from Authorization header
    if (!token) {
      const authHeader = request.headers.get('authorization');
      if (authHeader && authHeader.startsWith('Bearer ')) {
        token = authHeader.substring(7);
      }
    }
    
    // Try from request body (for some clients)
    if (!token && request.body) {
      const body = await request.json().catch(() => ({}));
      token = body.token;
    }
    
    if (!token) {
      return { user: null, error: 'No authentication token provided', status: 401 };
    }
    
    // Verify token
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    
    if (!decoded || !decoded.userId) {
      return { user: null, error: 'Invalid token', status: 401 };
    }
    
    // Find user (exclude password)
    const user = await User.findById(decoded.userId).select('-password');
    
    if (!user) {
      return { user: null, error: 'User not found', status: 404 };
    }
    
    if (user.isActive === false) {
      return { user: null, error: 'Account deactivated', status: 403 };
    }
    
    return { user, error: null };
    
  } catch (error) {
    console.error('Auth middleware error:', error);
    
    if (error.name === 'JsonWebTokenError') {
      return { user: null, error: 'Invalid token', status: 401 };
    }
    if (error.name === 'TokenExpiredError') {
      return { user: null, error: 'Token expired', status: 401 };
    }
    
    return { user: null, error: 'Authentication failed', status: 500 };
  }
}

export function withAuth(handler) {
  return async (request, context) => {
    const { user, error, status } = await authMiddleware(request);
    
    if (error || !user) {
      return new Response(
        JSON.stringify({ success: false, error: error || 'Unauthorized' }),
        { status: status || 401, headers: { 'Content-Type': 'application/json' } }
      );
    }
    
    return handler(request, { ...context, user });
  };
}

3. Create utils/phoneValidator.js
javascript
export function validatePakistanPhoneNumber(phone) {
  if (!phone) return true; // Allow empty
  // Format: +923XXXXXXXXX (3 for country code + 9 digits)
  const regex = /^\+923[0-9]{9}$/;
  return regex.test(phone);
}
