export function detectSeason(text) {
  const lower = text.toLowerCase();
  for (const [season, keywords] of Object.entries(SEASONAL_KEYWORDS)) {
    if (keywords.some(k => lower.includes(k))) {
      return season;
    }
  }
  return 'general';
}
