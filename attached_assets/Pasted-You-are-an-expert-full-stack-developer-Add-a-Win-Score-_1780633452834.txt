You are an expert full-stack developer. Add a Win Score Details page to TrendSpy that shows the complete breakdown of how a product's winning score is calculated. Use REAL data from the database – NO mock data.

## Goal

Users can click "View Score Breakdown" on any product and see:
- Total Win Score (0-100) with color-coded gauge
- Individual scores for each of the 7 signals
- Real data sources (Facebook ads count, TikTok views, Daraz orders, etc.)
- Why the product is winning (or not)

### 1. Create API Endpoint

Create `app/api/products/[slug]/score/route.js`:

```javascript
import { connectDB } from '@/lib/db';
import Product from '@/models/Product';
import ScrapedAd from '@/models/ScrapedAd';

export async function GET(request, { params }) {
  await connectDB();
  
  const product = await Product.findOne({ slug: params.slug });
  if (!product) {
    return Response.json({ success: false, error: 'Product not found' }, { status: 404 });
  }
  
  // Get real competitor count from Facebook ads
  const competitorCount = await ScrapedAd.countDocuments({
    category: product.category,
    scrapedAt: { $gte: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000) }
  });
  
  // Calculate individual scores (0-100 scale before weighting)
  const breakdown = {
    facebookAds: {
      score: Math.min(100, (product.activeAds || 0) * 2),
      weight: 20,
      rawValue: product.activeAds || 0,
      source: `${product.activeAds || 0} active ads on Facebook`
    },
    tiktok: {
      score: Math.min(100, Math.floor((product.tiktokViews || 0) / 100000)),
      weight: 15,
      rawValue: product.tiktokViews || 0,
      source: `${(product.tiktokViews || 0).toLocaleString()} views on TikTok`
    },
    daraz: {
      score: Math.min(100, Math.floor((product.darazOrders || 0) / 100)),
      weight: 20,
      rawValue: product.darazOrders || 0,
      source: `${(product.darazOrders || 0).toLocaleString()} orders on Daraz`
    },
    googleTrends: {
      score: product.googleTrendSpike || 0,
      weight: 15,
      rawValue: product.googleTrendSpike || 0,
      source: `${product.googleTrendSpike || 0}% spike in Google searches`
    },
    competitors: {
      score: Math.min(100, competitorCount),
      weight: 10,
      rawValue: competitorCount,
      source: `${competitorCount} competitors selling similar product`
    },
    seasonal: {
      score: product.seasonalRelevance || 50,
      weight: 10,
      rawValue: product.seasonalRelevance || 50,
      source: getSeasonalSource(product.category)
    },
    social: {
      score: Math.min(100, Math.floor((product.tiktokViews || 0) / 200000)),
      weight: 10,
      rawValue: product.tiktokViews || 0,
      source: `${(product.tiktokViews || 0).toLocaleString()} social engagements`
    }
  };
  
  // Calculate weighted total
  const totalScore = breakdown.facebookAds.score * 0.20 +
    breakdown.tiktok.score * 0.15 +
    breakdown.daraz.score * 0.20 +
    breakdown.googleTrends.score * 0.15 +
    breakdown.competitors.score * 0.10 +
    breakdown.seasonal.score * 0.10 +
    breakdown.social.score * 0.10;
  
  return Response.json({
    success: true,
    data: {
      productName: product.name,
      category: product.category,
      totalScore: Math.round(totalScore),
      breakdown,
      isWinning: Math.round(totalScore) >= 75,
      recommendation: Math.round(totalScore) >= 75 
        ? '🔥 Strong winning product – consider sourcing immediately'
        : Math.round(totalScore) >= 60
        ? '📈 Promising product – monitor closely'
        : '⚠️ Low confidence – wait for more signals'
    }
  });
}

function getSeasonalSource(category) {
  const seasonal = {
    'Electronics': 'High demand in Q4',
    'Fashion': 'Seasonal trends changing',
    'Beauty': 'Year-round steady demand',
    'Home': 'Peak during wedding season'
  };
  return seasonal[category] || 'Standard seasonal demand';
}

Create src/components/WinScoreDetails.jsx:

jsx
import { useState, useEffect } from 'react';
import { useAuthStore } from '@/store/useAuthStore';

export default function WinScoreDetails({ productSlug, onClose }) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const { token } = useAuthStore();
  
  useEffect(() => {
    fetchScore();
  }, [productSlug]);
  
  const fetchScore = async () => {
    try {
      const res = await fetch(`/api/products/${productSlug}/score`, {
        headers: { 'Authorization': `Bearer ${token}` }
      });
      const result = await res.json();
      if (result.success) setData(result.data);
    } catch (error) {
      console.error('Failed to fetch score:', error);
    } finally {
      setLoading(false);
    }
  };
  
  if (loading) return <div className="text-center py-8">Loading score breakdown...</div>;
  if (!data) return null;
  
  const signals = [
    { key: 'facebookAds', name: 'Facebook Ads', icon: '📱', color: 'blue' },
    { key: 'tiktok', name: 'TikTok', icon: '🎵', color: 'pink' },
    { key: 'daraz', name: 'Daraz Sales', icon: '🛒', color: 'orange' },
    { key: 'googleTrends', name: 'Google Trends', icon: '📈', color: 'green' },
    { key: 'competitors', name: 'Competitors', icon: '🏢', color: 'purple' },
    { key: 'seasonal', name: 'Seasonal', icon: '📅', color: 'yellow' },
    { key: 'social', name: 'Social Media', icon: '💬', color: 'red' }
  ];
  
  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50 p-4">
      <div className="bg-gray-800 rounded-xl max-w-2xl w-full max-h-[90vh] overflow-y-auto">
        {/* Header */}
        <div className="sticky top-0 bg-gray-800 border-b border-gray-700 p-4 flex justify-between items-center">
          <div>
            <h2 className="text-xl font-bold text-white">{data.productName}</h2>
            <p className="text-gray-400 text-sm">{data.category}</p>
          </div>
          <button onClick={onClose} className="text-gray-400 hover:text-white text-2xl">&times;</button>
        </div>
        
        {/* Total Score */}
        <div className="p-6 text-center border-b border-gray-700">
          <div className="inline-flex items-center justify-center w-32 h-32 rounded-full border-8 border-blue-500 mb-4">
            <span className="text-3xl font-bold text-white">{data.totalScore}</span>
          </div>
          <p className="text-white text-lg font-semibold">Total Win Score</p>
          <p className={`mt-2 text-sm ${data.isWinning ? 'text-green-400' : 'text-yellow-400'}`}>
            {data.recommendation}
          </p>
        </div>
        
        {/* Breakdown */}
        <div className="p-6">
          <h3 className="text-white font-semibold mb-4">Signal Breakdown</h3>
          <div className="space-y-4">
            {signals.map((signal) => {
              const item = data.breakdown[signal.key];
              if (!item) return null;
              const weightedScore = Math.round(item.score * (item.weight / 100));
              
              return (
                <div key={signal.key} className="bg-gray-700 rounded-lg p-3">
                  <div className="flex justify-between items-center mb-2">
                    <div className="flex items-center gap-2">
                      <span>{signal.icon}</span>
                      <span className="text-white font-medium">{signal.name}</span>
                      <span className="text-gray-400 text-xs">({item.weight}%)</span>
                    </div>
                    <span className="text-white font-bold">{weightedScore}/{item.weight}</span>
                  </div>
                  <div className="w-full bg-gray-600 rounded-full h-2">
                    <div className="bg-blue-500 h-2 rounded-full" style={{ width: `${item.score}%` }}></div>
                  </div>
                  <p className="text-gray-400 text-xs mt-1">{item.source}</p>
                </div>
              );
            })}
          </div>
        </div>
        
        {/* Close Button */}
        <div className="p-4 border-t border-gray-700">
          <button onClick={onClose} className="w-full bg-blue-600 hover:bg-blue-700 text-white py-2 rounded-lg">
            Close
          </button>
        </div>
      </div>
    </div>
  );
}
