### 1. Create `services/scraperService.js` – Production Scraper with Fallbacks

```javascript
import ScrapedAd from '@/models/ScrapedAd';
import { scrapeFacebookAds } from '@/scrapers/fbAdsScraper';

// In-memory cache for ads (5 minutes)
let adsCache = {
  data: null,
  lastFetch: null,
  loading: false
};

export async function getAdsWithFallback(options = {}) {
  const { forceRefresh = false, category = null, city = null, limit = 50 } = options;
  
  // Check if we have cached ads and not forcing refresh
  if (!forceRefresh && adsCache.data && adsCache.lastFetch) {
    const age = Date.now() - adsCache.lastFetch;
    if (age < 5 * 60 * 1000) { // 5 minutes cache
      return adsCache.data;
    }
  }
  
  // Prevent multiple concurrent scrapes
  if (adsCache.loading) {
    // Wait for existing scrape to finish
    await new Promise(resolve => setTimeout(resolve, 5000));
    return adsCache.data || [];
  }
  
  adsCache.loading = true;
  
  try {
    // Try live scraping with cookie
    const result = await scrapeFacebookAds(category || 'trending pakistan');
    
    if (result.success && result.ads && result.ads.length > 0) {
      // Save to database
      for (const ad of result.ads) {
        await ScrapedAd.findOneAndUpdate(
          { adId: ad.adId },
          { ...ad, scrapedAt: new Date() },
          { upsert: true }
        );
      }
      
      // Update cache
      adsCache.data = result.ads;
      adsCache.lastFetch = Date.now();
      adsCache.loading = false;
      
      return result.ads;
    }
    
    // If live scrape returns nothing, fallback to database
    console.log('⚠️ Live scrape returned 0 ads. Falling back to database...');
    const dbAds = await ScrapedAd.find({})
      .sort({ scrapedAt: -1 })
      .limit(limit);
    
    if (dbAds.length > 0) {
      adsCache.data = dbAds;
      adsCache.lastFetch = Date.now();
      adsCache.loading = false;
      return dbAds;
    }
    
    // If database is empty, return empty array
    adsCache.loading = false;
    return [];
    
  } catch (error) {
    console.error('❌ Scraper error:', error.message);
    
    // Fallback to database on error
    try {
      const dbAds = await ScrapedAd.find({})
        .sort({ scrapedAt: -1 })
        .limit(limit);
      
      if (dbAds.length > 0) {
        adsCache.data = dbAds;
        adsCache.lastFetch = Date.now();
        adsCache.loading = false;
        return dbAds;
      }
    } catch (dbError) {
      console.error('❌ Database fallback error:', dbError.message);
    }
    
    adsCache.loading = false;
    return [];
  }
}

Create app/api/scraper/health/route.js:

javascript
import ScrapedAd from '@/models/ScrapedAd';

export async function GET() {
  try {
    const count = await ScrapedAd.countDocuments();
    const latest = await ScrapedAd.findOne().sort({ scrapedAt: -1 });
    
    return Response.json({
      success: true,
      status: {
        totalAds: count,
        lastScraped: latest?.scrapedAt || null,
        hasCookie: !!process.env.FB_SESSION_COOKIE,
        healthy: count > 0 && latest?.scrapedAt > new Date(Date.now() - 7 * 24 * 60 * 60 * 1000)
      }
    });
  } catch (error) {
    return Response.json({ success: false, error: error.message }, { status: 500 });
  }
}
