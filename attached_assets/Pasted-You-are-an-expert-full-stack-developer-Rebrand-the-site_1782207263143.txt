1. Create src/context/ThemeContext.jsx
javascript
import { createContext, useContext, useState, useEffect } from 'react';

const ThemeContext = createContext();

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState(() => {
    // Check localStorage first, then default to 'light'
    const saved = localStorage.getItem('hunting-goals-theme');
    return saved || 'light';
  });

  useEffect(() => {
    localStorage.setItem('hunting-goals-theme', theme);
    document.documentElement.setAttribute('data-theme', theme);
  }, [theme]);

  const toggleTheme = () => {
    setTheme(prev => prev === 'light' ? 'dark' : 'light');
  };

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider');
  }
  return context;
}

4. Create src/components/ThemeToggle.jsx
jsx
import { useTheme } from '@/context/ThemeContext';
import { FiSun, FiMoon } from 'react-icons/fi';

export default function ThemeToggle({ variant = 'navbar' }) {
  const { theme, toggleTheme } = useTheme();

  const buttonClass = variant === 'navbar' 
    ? 'p-2 rounded-lg hover:bg-gray-100 dark:hover:bg-gray-700 transition text-gray-600 dark:text-gray-300'
    : 'w-full flex items-center gap-3 px-3 py-2 rounded-lg hover:bg-gray-100 dark:hover:bg-gray-700 transition text-gray-600 dark:text-gray-300';

  return (
    <button onClick={toggleTheme} className={buttonClass} title="Toggle theme">
      {theme === 'light' ? <FiMoon size={20} /> : <FiSun size={20} />}
      {variant === 'sidebar' && (
        <span>{theme === 'light' ? 'Dark Mode' : 'Light Mode'}</span>
      )}
    </button>
  );
}
