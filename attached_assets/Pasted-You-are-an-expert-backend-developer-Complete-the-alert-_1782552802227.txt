### 1. Update `services/alertService.js`

```javascript
import { sendWhatsAppAlert } from './whatsappService';
import { sendEmailAlert } from './emailService';
import AlertLog from '@/models/AlertLog';

export async function checkAndTriggerAlerts(product) {
  // Find matching active alerts
  const alerts = await Alert.find({
    city: { $in: product.cities },
    category: product.category,
    minWinScore: { $lte: product.winScore },
    isActive: true
  }).populate('userId');

  for (const alert of alerts) {
    const user = alert.userId;
    
    // Skip if user has notifications disabled
    if (!user.emailNotifications && !user.whatsappNotifications) continue;

    // Send Email
    if ((alert.channel === 'email' || alert.channel === 'both') && user.emailNotifications) {
      await sendEmailAlert(user.email, product);
    }

    // Send WhatsApp (with ad link)
    if ((alert.channel === 'whatsapp' || alert.channel === 'both') && user.whatsappNotifications) {
      if (user.phoneNumber) {
        await sendWhatsAppAlert(user.phoneNumber, product);
      }
    }

    // Log the alert (for history)
    await AlertLog.create({
      userId: user._id,
      alertId: alert._id,
      productId: product._id,
      productName: product.name,
      winScore: product.winScore,
      channel: alert.channel,
      sentAt: new Date(),
      delivered: true
    });

    // Update alert trigger count
    await Alert.findByIdAndUpdate(alert._id, {
      $inc: { triggerCount: 1 },
      lastTriggeredAt: new Date()
    });
  }
}

import { checkAllProductsForAlerts } from '@/services/alertService';

export async function startAlertJob() {
  // Only run if alerts are enabled
  if (process.env.ALERTS_ENABLED !== 'true') {
    console.log('⏸️ Alerts disabled. Set ALERTS_ENABLED=true to enable.');
    return;
  }

  try {
    await checkAllProductsForAlerts();
    console.log(`✅ Alert job completed at ${new Date().toISOString()}`);
  } catch (error) {
    console.error('❌ Alert job failed:', error.message);
  }
}
