### 1. Create API Endpoint

Create `app/api/suppliers/match/route.js`:

```javascript
import { connectDB } from '@/lib/db';
import Supplier from '@/models/Supplier';

export async function GET(request) {
  try {
    await connectDB();
    
    const { searchParams } = new URL(request.url);
    const category = searchParams.get('category');
    const city = searchParams.get('city');
    const productName = searchParams.get('productName');
    
    let query = {};
    
    // Match by category or product keywords
    if (category) {
      query.category = { $regex: new RegExp(category, 'i') };
    }
    
    // Match by city (preferred city first)
    if (city) {
      query.city = { $regex: new RegExp(city, 'i') };
    }
    
    // Also search by product name keywords
    if (productName) {
      const keywords = productName.split(' ').slice(0, 3);
      query.$or = [
        { products: { $in: keywords } },
        { name: { $regex: keywords.join('|'), $options: 'i' } }
      ];
    }
    
    // Get suppliers: verified first, then by rating
    const suppliers = await Supplier.find(query)
      .sort({ verificationStatus: -1, rating: -1 })
      .limit(5);
    
    // If no suppliers found by city, try major cities
    if (suppliers.length === 0 && city) {
      const fallbackSuppliers = await Supplier.find({
        city: { $in: ['Lahore', 'Karachi', 'Islamabad'] }
      }).limit(3);
      
      return Response.json({
        success: true,
        suppliers: fallbackSuppliers,
        fallbackUsed: true,
        message: `No suppliers found in ${city}. Showing major city suppliers.`
      });
    }
    
    return Response.json({
      success: true,
      suppliers,
      fallbackUsed: false
    });
    
  } catch (error) {
    console.error('Supplier match error:', error);
    return Response.json(
      { success: false, error: error.message, suppliers: [] },
      { status: 500 }
    );
  }
}

  if (productSlug && data) {
    fetchSuppliers();

const fetchSuppliers = async () => {
  setLoadingSuppliers(true);
  try {
    const res = await fetch(`/api/suppliers/match?category=${data?.category}&city=${data?.city || 'Lahore'}&productName=${data?.productName}`);
    const result = await res.json();
    if (result.success) {
      setSuppliers(result.suppliers);
    }
  } catch (error) {
    console.error('Failed to fetch suppliers:', error);
  } finally {
    setLoadingSuppliers(false);
  }
};
