import puppeteer from 'puppeteer-extra';
import StealthPlugin from 'puppeteer-extra-stealth';

puppeteer.use(StealthPlugin());

export async function scrapeFacebookAds(searchTerm, category = null) {
  let browser = null;
  const ads = [];
  
  try {
    const cookieString = process.env.FB_SESSION_COOKIE;
    
    if (!cookieString) {
      console.log('❌ FB_SESSION_COOKIE not set. Cannot scrape real ads.');
      return { success: false, ads: [], totalFound: 0, savedNew: 0, error: 'Cookie missing' };
    }
    
    browser = await puppeteer.launch({
      headless: 'new',
      args: ['--no-sandbox', '--disable-setuid-sandbox']
    });
    
    const page = await browser.newPage();
    
    // Parse and set cookies
    const cookies = cookieString.split('; ').map(cookie => {
      const [name, value] = cookie.split('=');
      return { name, value, domain: '.facebook.com' };
    });
    
    await page.setCookie(...cookies);
    await page.setUserAgent('Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36');
    
    // Facebook Ad Library URL for Pakistan
    const url = `https://www.facebook.com/ads/library/?active_status=all&ad_type=all&country=PK&q=${encodeURIComponent(searchTerm)}`;
    
    await page.goto(url, { waitUntil: 'networkidle2', timeout: 30000 });
    await page.waitForSelector('div[data-ad-preview]', { timeout: 15000 });
    
    // Extract ads
    const extractedAds = await page.evaluate(() => {
      const adElements = document.querySelectorAll('div[data-ad-preview]');
      return Array.from(adElements).slice(0, 20).map(ad => {
        const adId = ad.getAttribute('data-ad-id') || 
                     ad.querySelector('a[href*="ads/library"]')?.href?.match(/id=(\d+)/)?.[1];
        
        return {
          adId: adId,
          directUrl: adId ? `https://www.facebook.com/ads/library/?id=${adId}` : null,
          advertiserName: ad.querySelector('[data-advertiser-name]')?.innerText || 
                          ad.querySelector('strong')?.innerText,
          headline: ad.querySelector('[data-ad-body]')?.innerText ||
                    ad.querySelector('div[dir="auto"]')?.innerText,
          daysRunning: parseInt(ad.querySelector('[data-running-days]')?.innerText) || null,
          creativeType: ad.querySelector('video') ? 'video' : 
                        ad.querySelector('[data-carousel]') ? 'carousel' : 'image',
          platform: 'facebook'
        };
      });
    });
    
    ads.push(...extractedAds);
    await browser.close();
    
    // Save to database
    let savedNew = 0;
    for (const ad of ads) {
      if (ad.adId) {
        const existing = await ScrapedAd.findOne({ adId: ad.adId });
        if (!existing) {
          await ScrapedAd.create({
            adId: ad.adId,
            directUrl: ad.directUrl,
            advertiserName: ad.advertiserName,
            headline: ad.headline,
            daysRunning: ad.daysRunning,
            creativeType: ad.creativeType,
            platform: ad.platform,
            category: category,
            scrapedAt: new Date()
          });
          savedNew++;
        }
      }
    }
    
    return { success: true, ads, totalFound: ads.length, savedNew };
    
  } catch (error) {
    console.error('FB Scraper error:', error);
    if (browser) await browser.close();
    return { success: false, ads: [], totalFound: 0, savedNew: 0, error: error.message };
  }
}

export async function countCompetitors(productName) {
  const result = await scrapeFacebookAds(productName);
  const uniqueAdvertisers = new Set(result.ads.map(ad => ad.advertiserName));
  return { count: uniqueAdvertisers.size, advertisers: Array.from(uniqueAdvertisers) };
}

import { scrapeFacebookAds } from '@/scrapers/fbAdsScraper';
import { emitNewAdsDetected } from '@/lib/socketEmitter';

export async function startFbAdsJob() {
  const categories = ['Electronics', 'Fashion', 'Beauty', 'Home', 'Sports'];
  let totalSaved = 0;
  
  for (const category of categories) {
    const result = await scrapeFacebookAds(category, category);
    if (result.savedNew > 0) {
      totalSaved += result.savedNew;
      emitNewAdsDetected(category, result.savedNew);
    }
    await new Promise(resolve => setTimeout(resolve, 5000)); // delay
  }
  
  console.log(`✅ FB Ads job completed. New ads: ${totalSaved}`);
}
