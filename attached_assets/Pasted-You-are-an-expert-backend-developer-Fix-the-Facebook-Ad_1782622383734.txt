### 1. Update `scrapers/fbAdsScraper.js` — Extract Real adId

```javascript
// In the scraper, when extracting ad data:
function extractRealAdId(element) {
  // Method 1: Check data-ad-id attribute (numeric only)
  const adIdAttr = element.getAttribute('data-ad-id');
  if (adIdAttr && /^[0-9]+$/.test(adIdAttr)) {
    return adIdAttr;
  }
  
  // Method 2: Check for ad_archive_id in any data attribute
  const allAttrs = element.attributes;
  for (const attr of allAttrs) {
    if (attr.name.includes('ad') && attr.value && /^[0-9]{10,}$/.test(attr.value)) {
      return attr.value;
    }
  }
  
  // Method 3: Extract from internal JSON in the page
  const pageHTML = document.documentElement.innerHTML;
  const match = pageHTML.match(/"ad_archive_id":"(\d+)"/);
  if (match) return match[1];
  
  // Method 4: Extract from any link containing numeric ID
  const links = element.querySelectorAll('a[href*="ads/library"]');
  for (const link of links) {
    const match = link.href.match(/id=(\d+)/);
    if (match && /^[0-9]+$/.test(match[1])) {
      return match[1];
    }
  }
  
  return null; // Don't save fake IDs
}
